<script>
const $=id=>document.getElementById(id);
const TYPES=['R','V','I','VCVS','VCCS','CCVS','CCCS'];
const f=x=>String(+Number(x).toPrecision(5));
const MULT={G:1e9,k:1e3,M:1e6,m:1e-3,u:1e-6,n:1e-9,p:1e-12};

function num(s){
  const m=/^\s*([-+]?(?:\d+\.?\d*|\.\d+)(?:[eE][-+]?\d+)?)\s*([GkMmunp]?)\s*$/.exec(s);
  if(!m) throw new Error('Bad value: "'+s+'"');
  return parseFloat(m[1])*(m[2]?MULT[m[2]]:1);
}

function addRow(c){
  c=c||{};
  const tr=document.createElement('tr');
  tr.innerHTML='<td><input class="nm" placeholder="Name" aria-label="Name"></td><td><select class="ty" aria-label="Type">'+TYPES.map(t=>'<option>'+t+'</option>').join('')+'</select></td>'+
    '<td><input class="np" placeholder="Node +" aria-label="Node +"></td><td><input class="nn" placeholder="Node -" aria-label="Node -"></td><td><input class="vl" placeholder="Value" aria-label="Value"></td><td><input class="ct" placeholder="Control" aria-label="Control"></td>'+
    '<td><button class="g" aria-label="Delete row">x</button></td>';
  tr.querySelector('.nm').value=c.name||'';tr.querySelector('.ty').value=c.type||'R';
  tr.querySelector('.np').value=c.p||'';tr.querySelector('.nn').value=c.n||'';
  tr.querySelector('.vl').value=c.val||'';tr.querySelector('.ct').value=c.ctl||'';
  tr.querySelector('button').onclick=()=>tr.remove();
  $('tbl').tBodies[0].appendChild(tr);
}

function read(){
  const els=[],seen={};
  for(const tr of $('tbl').tBodies[0].rows){
    const q=s=>tr.querySelector(s).value.trim();
    const e={name:q('.nm'),type:q('.ty'),p:q('.np'),n:q('.nn'),ctl:q('.ct')};
    if(!e.name) throw new Error('Every element needs a name.');
    if(seen[e.name]) throw new Error('Duplicate name: '+e.name);
    seen[e.name]=1;
    if(!e.p||!e.n) throw new Error(e.name+': both nodes are required.');
    const vs=q('.vl');e.val=(vs===''||vs==='?')?null:num(vs);
    if(e.type==='R'&&e.val===0) throw new Error(e.name+': resistance cannot be 0. Use a 0 V source for a wire.');
    els.push(e);
  }
  if(!els.length) throw new Error('Add at least one element.');
  return els;
}

function solve(els){
  for(const e of els) if(e.val===null) throw new Error(e.name+' has no value. Enter one, or choose it as the load.');
  const nodes=[...new Set(els.flatMap(e=>[e.p,e.n]).filter(x=>x!=='0'))];
  const ni={};nodes.forEach((n,i)=>ni[n]=i);
  const ix=n=>n==='0'?-1:ni[n];
  const br=els.filter(e=>e.type==='V'||e.type==='VCVS'||e.type==='CCVS');
  const bi={};br.forEach((e,k)=>bi[e.name]=nodes.length+k);
  const N=nodes.length+br.length;
  const A=Array.from({length:N},()=>Array(N).fill(0)),b=Array(N).fill(0);
  const add=(r,c,v)=>{if(r>=0&&c>=0)A[r][c]+=v};
  const byName={};els.forEach(e=>byName[e.name]=e);

  // control current as list of [column, coefficient]
  const ctlI=e=>{
    const c=byName[e.ctl];
    if(!c) throw new Error(e.name+': control element "'+e.ctl+'" not found.');
    if(c.type==='R') return [[ix(c.p),1/c.val],[ix(c.n),-1/c.val]];
    if(bi[c.name]!==undefined&&c.type==='V') return [[bi[c.name],1]];
    throw new Error(e.name+': control must be a resistor or a V source.');
  };
  const ctlV=e=>{
    const t=e.ctl.split(',').map(s=>s.trim());
    if(t.length!==2) throw new Error(e.name+': control needs two nodes like 2,0');
    return t.map(ix);
  };

  for(const e of els){
    const p=ix(e.p),n=ix(e.n),v=e.val,k=bi[e.name];
    if(p===undefined||n===undefined) throw new Error('Bad node in '+e.name);
    if(e.type==='R'){const g=1/v;add(p,p,g);add(n,n,g);add(p,n,-g);add(n,p,-g);}
    else if(e.type==='I'){if(p>=0)b[p]-=v;if(n>=0)b[n]+=v;}
    else if(e.type==='VCCS'){const [c1,c2]=ctlV(e);add(p,c1,v);add(p,c2,-v);add(n,c1,-v);add(n,c2,v);}
    else if(e.type==='CCCS'){for(const [c,co] of ctlI(e)){add(p,c,v*co);add(n,c,-v*co);}}
    else{
      add(p,k,1);add(n,k,-1);add(k,p,1);add(k,n,-1);
      if(e.type==='V') b[k]=v;
      else if(e.type==='VCVS'){const [c1,c2]=ctlV(e);add(k,c1,-v);add(k,c2,v);}
      else for(const [c,co] of ctlI(e)) add(k,c,-v*co);
    }
  }

  // working text
  const vname=c=>c<nodes.length?'V'+nodes[c]:'i('+br[c-nodes.length].name+')';
  const eqs=A.map((row,r)=>{
    const terms=row.map((a,c)=>a?[a,c]:null).filter(Boolean)
      .map(([a,c],i)=>(a<0?'- ':(i?'+ ':''))+(Math.abs(a)===1?'':f(Math.abs(a))+' ')+vname(c)).join(' ');
    const lab=r<nodes.length?'KCL node '+nodes[r]:'Branch '+br[r-nodes.length].name;
    return lab+':  '+(terms||'0')+' = '+f(b[r]);
  });

  // Gaussian elimination, partial pivoting
  const M=A.map((row,i)=>[...row,b[i]]);
  for(let c=0;c<N;c++){
    let piv=c;for(let r=c+1;r<N;r++) if(Math.abs(M[r][c])>Math.abs(M[piv][c])) piv=r;
    if(Math.abs(M[piv][c])<1e-12) throw new Error('Cannot solve: floating node, loop of voltage sources, or no ground (node 0).');
    [M[c],M[piv]]=[M[piv],M[c]];
    for(let r=0;r<N;r++){
      if(r===c) continue;
      const m=M[r][c]/M[c][c];
      if(m) for(let k=c;k<=N;k++) M[r][k]-=m*M[c][k];
    }
  }
  const x=M.map((row,i)=>row[N]/row[i]);
  const V=n=>{
    if(n==='0') return 0;
    if(ni[n]===undefined) throw new Error('Node "'+n+'" is not in the circuit.');
    return x[ni[n]];
  };
  const rows=els.map(e=>{
    const vd=V(e.p)-V(e.n);let i;
    if(e.type==='R') i=vd/e.val;
    else if(e.type==='I') i=e.val;
    else if(e.type==='VCCS'){const [c1,c2]=ctlV(e);const g=c=>c<0?0:x[c];i=e.val*(g(c1)-g(c2));}
    else if(e.type==='CCCS') i=e.val*ctlI(e).reduce((s,[c,co])=>s+(c<0?0:co*x[c]),0);
    else i=x[bi[e.name]];
    return {e,vd,i,P:vd*i};
  });
  return {nodes,V,rows,eqs};
}

function report(s){
  let t='Equations (matrix rows):\n'+s.eqs.join('\n')+'\n\nNode voltages:\n';
  t+=s.nodes.map(n=>'V'+n+' = '+f(s.V(n))+' V').join('\n');
  t+='\n\nElements (i from + to - through element, P absorbed):\n';
  t+=s.rows.map(r=>(r.e.name+'  v = '+f(r.vd)+' V   i = '+f(r.i)+' A   P = '+f(r.P)+' W')).join('\n');
  t+='\n\nPower check: sum of P = '+f(s.rows.reduce((a,r)=>a+r.P,0))+' W (should be 0)';
  return t;
}

function run(fn){
  try{$('out').className='';$('out').textContent=fn();}
  catch(e){$('out').className='err';$('out').textContent=e.message;}
}

function lin(A,b){
  const N=b.length,M=A.map((r,i)=>[...r,b[i]]);
  for(let c=0;c<N;c++){
    let p=c;for(let r=c+1;r<N;r++) if(Math.abs(M[r][c])>Math.abs(M[p][c])) p=r;
    if(Math.abs(M[p][c])<1e-12) throw new Error('Mesh equations cannot be solved. Check each mesh is a closed loop and shared elements appear in both meshes.');
    [M[c],M[p]]=[M[p],M[c]];
    for(let r=0;r<N;r++){if(r===c)continue;const m=M[r][c]/M[c][c];if(m)for(let k=c;k<=N;k++)M[r][k]-=m*M[c][k];}
  }
  return M.map((r,i)=>r[N]/r[i]);
}

function meshes(){
  const els=read(),by={};els.forEach(e=>by[e.name]=e);
  const lines=$('mesh').value.split('\n').map(s=>s.trim()).filter(Boolean);
  if(!lines.length) throw new Error('Describe at least one mesh.');
  const K=lines.length;
  const S=lines.map(l=>{
    const o={};
    l.replace(/^[^:]*:/,'').trim().split(/\s+/).forEach(t=>{
      const sg=t.endsWith('-')?-1:1,nm=t.replace(/[+-]$/,''),e=by[nm];
      if(!e) throw new Error('Unknown element: '+nm);
      if(!['R','V','I'].includes(e.type)) throw new Error(nm+': mesh analysis here supports R, V and I only.');
      o[nm]=sg;
    });
    return o;
  });
  // KVL row per mesh: sum of drops = 0, current sources left out
  const rows=[];
  for(let k=0;k<K;k++){
    const c=Array(K).fill(0);let r=0;
    for(const nm in S[k]){
      const e=by[nm],s=S[k][nm];
      if(e.type==='R'){for(let j=0;j<K;j++) if(S[j][nm]) c[j]+=s*S[j][nm]*e.val;}
      else if(e.type==='V') r-=s*e.val;
    }
    rows.push({c,r,m:[k],on:true});
  }
  const rep=[...Array(K).keys()],find=x=>{while(rep[x]!==x)x=rep[x];return x};
  const cons=[];
  for(const e of els.filter(e=>e.type==='I')){
    const ms=[];for(let j=0;j<K;j++) if(S[j][e.name]) ms.push(j);
    if(!ms.length) continue;
    if(ms.length>2) throw new Error(e.name+' is in more than 2 meshes.');
    const c=Array(K).fill(0);ms.forEach(j=>c[j]=S[j][e.name]);
    cons.push({c,r:e.val,src:e.name});
    const ra=find(ms[0]);
    if(ms.length===1) rows[ra].on=false;           // mesh current is fixed
    else{                                          // supermesh: add the two KVL rows
      const rb=find(ms[1]);
      if(ra!==rb){
        rows[ra].c=rows[ra].c.map((v,i)=>v+rows[rb].c[i]);
        rows[ra].r+=rows[rb].r;rows[ra].m=rows[ra].m.concat(rows[rb].m);
        rows[rb].on=false;rep[rb]=ra;
      }
    }
  }
  const eq=rows.filter(r=>r.on).map(r=>({...r,l:r.m.length>1?'Supermesh (i'+r.m.map(j=>j+1).join(', i')+')':'KVL mesh '+(r.m[0]+1)}))
    .concat(cons.map(c=>({...c,l:'Current source '+c.src})));
  if(eq.length!==K) throw new Error('Mesh setup gives '+eq.length+' equations for '+K+' meshes. Check that each mesh is a closed loop.');
  const txt=eq.map(q=>q.l+':  '+(q.c.map((a,j)=>a?[a,j]:null).filter(Boolean)

}

// Fallback auto-layout (used only when the photo reader gave no positions)
function relax(els,nodes,move){
  const mv=new Set(move),k=115;
  const edges=els.filter(e=>e.p!=='0'&&e.n!=='0');
  for(let it=0;it<300;it++){
    const fo={};nodes.forEach(n=>fo[n]={x:0,y:0});
    for(let i=0;i<nodes.length;i++)for(let j=i+1;j<nodes.length;j++){
      const a=POS[nodes[i]],b=POS[nodes[j]];
      let dx=a.x-b.x,dy=a.y-b.y;const d=Math.hypot(dx,dy)||0.01,r=k*k/d;dx/=d;dy/=d;
      fo[nodes[i]].x+=dx*r;fo[nodes[i]].y+=dy*r;fo[nodes[j]].x-=dx*r;fo[nodes[j]].y-=dy*r;
    }
    for(const e of edges){
      const a=POS[e.p],b=POS[e.n];
      let dx=b.x-a.x,dy=b.y-a.y;const d=Math.hypot(dx,dy)||0.01,at=d*d/k;dx/=d;dy/=d;
      fo[e.p].x+=dx*at;fo[e.p].y+=dy*at;fo[e.n].x-=dx*at;fo[e.n].y-=dy*at;
    }
    const xs=nodes.map(n=>POS[n].x),minx=Math.min(...xs),maxx=Math.max(...xs),span=Math.max(1,maxx-minx);
    for(const n of nodes){
      const u=(POS[n].x-minx)/span;
      fo[n].x+=(60+u*(SVGW-120)-POS[n].x)*0.04;
      fo[n].y+=((GY-130)/2+40-POS[n].y)*0.05;
    }
    const t=30*(1-it/300)+1;
    for(const n of mv){
      const m=Math.hypot(fo[n].x,fo[n].y)||1,sc=Math.min(m,t)/m;
      POS[n]={x:Math.max(40,Math.min(SVGW-40,POS[n].x+fo[n].x*sc)),y:Math.max(40,Math.min(GY-70,POS[n].y+fo[n].y*sc))};
    }
  }
}

// Positions read from the photo: {"1":[col,row], ...}, row 0 = top, ground node 0 not needed
function applyLayout(nd){
  POS={};
  const ks=Object.keys(nd).filter(k=>k!=='0'&&Array.isArray(nd[k]));
  if(!ks.length) return;
  const cs=ks.map(k=>+nd[k][0]),rs=ks.map(k=>+nd[k][1]);
  const c0=Math.min(...cs),c1=Math.max(...cs),r0=Math.min(...rs),r1=Math.max(...rs);
  const W=SVGW-200,H=GY-70-60;
  ks.forEach(k=>{

  });
}

// one element symbol along pts (pts[0] = + terminal). Returns SVG string.
function sym(e,pts,ann){
  let h='',bi=0,bl=-1;
  for(let i=0;i<pts.length-1;i++){
    const l=Math.hypot(pts[i+1].x-pts[i].x,pts[i+1].y-pts[i].y);
    if(l>bl){bl=l;bi=i;}
  }
  const a=pts[bi],b=pts[bi+1],D=bl||1,ux=(b.x-a.x)/D,uy=(b.y-a.y)/D;
  const M={x:(a.x+b.x)/2,y:(a.y+b.y)/2},L=Math.max(24,Math.min(LSYM,D-4)),sc=L/LSYM;
  let d='';
  for(let i=0;i<pts.length-1;i++){
    const p=pts[i],q=pts[i+1];
    if(i===bi) d+='M'+p.x+' '+p.y+'L'+(M.x-ux*L/2)+' '+(M.y-uy*L/2)+'M'+(M.x+ux*L/2)+' '+(M.y+uy*L/2)+'L'+q.x+' '+q.y;
    else d+='M'+p.x+' '+p.y+'L'+q.x+' '+q.y;
  }
  h+='<path class="el" d="'+d+'"/>';
  h+='<g transform="translate('+M.x+' '+M.y+') rotate('+(Math.atan2(uy,ux)*180/Math.PI)+') scale('+sc+')">'+SHAPE[e.type]+'</g>';
  if(VOLT.includes(e.type)){
    h+='<text x="'+(M.x-ux*5)+'" y="'+(M.y-uy*5+4)+'" text-anchor="middle">+</text><text x="'+(M.x+ux*5)+'" y="'+(M.y+uy*5+4)+'" text-anchor="middle">\u2212</text>';
  }
  let nx=-uy,ny=ux;if(ny>0.001||(Math.abs(ny)<=0.001&&nx<0)){nx=-nx;ny=-ny;}
  const lines=[e.name,unitOf(e)];if(ann&&ann.I[e.name]!==undefined) lines.push('i = '+f(ann.I[e.name])+' A');
  const lx=M.x+nx*26,ly=M.y+ny*26,anc=nx>0.5?'start':nx<-0.5?'end':'middle';
  const y0=anc==='middle'?ly-13*(lines.length-1):ly-6*(lines.length-1);
  h+='<text text-anchor="'+anc+'" x="'+lx+'" y="'+y0+'">'+lines.map((l,i)=>'<tspan x="'+lx+'" dy="'+(i?13:0)+'"'+(i===0?' class="nm"':'')+'>'+esc(l)+'</tspan>').join('')+'</text>';

function drawGeo(svg,els,ann){
  svg.setAttribute('viewBox','0 0 '+SVGW+' '+SVGH);
  const all=[];
  els.forEach(e=>all.push(GEO.el[e.name].a,GEO.el[e.name].b));
  GEO.wires.forEach(w=>all.push(w[0],w[1]));
  const uniq=i=>[...new Set(all.map(p=>p[i]))].sort((a,b)=>a-b);
  const C=uniq(0),R=uniq(1);
  const sx=C.length>1?Math.min(230,(SVGW-200)/(C.length-1)):0;
  const sy=R.length>1?Math.min(150,(GY-110)/(R.length-1)):0;
  const x0=(SVGW-sx*(C.length-1))/2,y0=60;
  const T=p=>({x:x0+C.indexOf(p[0])*sx,y:y0+R.indexOf(p[1])*sy});
  const key=p=>p[0]+','+p[1];

  const nodeAt={},deg={},inc=p=>{deg[key(p)]=(deg[key(p)]||0)+1;};
  els.forEach(e=>{const g=GEO.el[e.name];nodeAt[key(g.a)]=e.p;nodeAt[key(g.b)]=e.n;inc(g.a);inc(g.b);});
  GEO.wires.forEach(w=>{inc(w[0]);inc(w[1]);});

  let h='';
  GEO.wires.forEach(w=>{const p=T(w[0]),q=T(w[1]);h+='<path class="el" d="M'+p.x+' '+p.y+'L'+q.x+' '+q.y+'"/>';});
  els.forEach(e=>{const g=GEO.el[e.name];h+=sym(e,[T(g.a),T(g.b)],ann);});

  // junction dots
  const pts={};all.forEach(p=>pts[key(p)]=p);
  for(const k in pts) if((deg[k]||0)>=3){const t=T(pts[k]);h+='<circle class="dot" cx="'+t.x+'" cy="'+t.y+'" r="3.5"/>';}

  // one label per node (top-left-most point of that node)
  const best={};
  for(const k in nodeAt){
    const n=nodeAt[k];if(n==='0')continue;
    const p=pts[k];if(!best[n]||p[1]<best[n][1]||(p[1]===best[n][1]&&p[0]<best[n][0])) best[n]=p;
  }
  for(const n in best){
    const t=T(best[n]),lab=ann&&ann.V[n]!==undefined?'V'+n+' = '+f(ann.V[n])+' V':n;
    h+='<circle class="dot" cx="'+t.x+'" cy="'+t.y+'" r="4.5"/><text class="nm" x="'+(t.x+7)+'" y="'+(t.y-8)+'">'+esc(lab)+'</text>';
  }

  // ground symbol under the lowest ground point
  const g0=Object.keys(nodeAt).filter(k=>nodeAt[k]==='0').map(k=>T(pts[k]));
  if(g0.length){
    const gy=Math.max(...g0.map(p=>p.y)),row=g0.filter(p=>p.y===gy);
    const gx=(Math.min(...row.map(p=>p.x))+Math.max(...row.map(p=>p.x)))/2;
    h+='<path class="el" d="M'+gx+' '+gy+'v8M'+(gx-12)+' '+(gy+8)+'h24M'+(gx-7)+' '+(gy+13)+'h14M'+(gx-3)+' '+(gy+18)+'h6"/>';
  }
  svg.innerHTML=h;
}

function draw(svg,els,ann){

    });

    fresh.forEach(n=>{POS[n]={x:snap(POS[n].x),y:snap(POS[n].y)};});
  }
  const grp={},gnd={};
  els.forEach(e=>{
    if(e.p==='0'||e.n==='0'){const o=e.p==='0'?e.n:e.p;(gnd[o]=gnd[o]||[]).push(e);}
    else{const key=[e.p,e.n].sort().join('|');(grp[key]=grp[key]||[]).push(e);}
  });
  let h='';

  // pts runs from the element's + terminal (node p) to its - terminal (node n), all segments horizontal or vertical
  const put=(e,pts)=>{h+=sym(e,pts,ann);};

  // elements between two non-ground nodes: straight if aligned, otherwise an L-shaped wire
  for(const key in grp){
    const list=grp[key];
    list.forEach((e,k)=>{
      const A=POS[e.p],B=POS[e.n];
      let pts;
      if(A.x===B.x||A.y===B.y){
        const off=(k-(list.length-1)/2)*GRID*2;
        if(off===0) pts=[A,B];
        else{
          const dx=B.x-A.x,dy=B.y-A.y,D=Math.hypot(dx,dy)||1,ox=-dy/D*off,oy=dx/D*off;
          pts=[A,{x:A.x+ox,y:A.y+oy},{x:B.x+ox,y:B.y+oy},B];
        }
      }else{
        pts=[A,k%2?{x:A.x,y:B.y}:{x:B.x,y:A.y},B];
      }
      put(e,pts);
    });
  }

  // elements to ground drop straight down to one shared ground rail
  const gxs=[];
  for(const o in gnd){
    gnd[o].forEach((e,k)=>{
      const P=POS[o],off=(k-(gnd[o].length-1)/2)*60,G={x:P.x+off,y:GY};
      const pts=off===0?[P,G]:[P,{x:P.x+off,y:P.y},G];
      if(e.p==='0') pts.reverse();
      gxs.push(G.x);put(e,pts);
    });
  }
  if(gxs.length){
    const x0=Math.min(...gxs),x1=Math.max(...gxs),gm=(x0+x1)/2;
    h+='<path class="el" d="M'+x0+' '+GY+'H'+x1+'M'+gm+' '+GY+'v8M'+(gm-12)+' '+(GY+8)+'h24M'+(gm-7)+' '+(GY+13)+'h14M'+(gm-3)+' '+(GY+18)+'h6"/>';
  }

  for(const n of nodes){
    const P=POS[n],lab=ann&&ann.V[n]!==undefined?'V'+n+' = '+f(ann.V[n])+' V':n;
    h+='<circle class="dot" cx="'+P.x+'" cy="'+P.y+'" r="4.5"/><text class="nm" x="'+(P.x+7)+'" y="'+(P.y-8)+'">'+esc(lab)+'</text>';
    if(!ann) h+='<circle class="hit" data-n="'+esc(n)+'" cx="'+P.x+'" cy="'+P.y+'" r="15"/>';
  }
  svg.innerHTML=h;
}
